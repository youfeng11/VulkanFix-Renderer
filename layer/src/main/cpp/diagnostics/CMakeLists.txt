# device cannot initialize the wrapper at all.

target_compile_features(vulkanfix_driver_post PRIVATE cxx_std_17)
target_include_directories(vulkanfix_driver_post PRIVATE ../../../../../Vulkan-Headers/include)
target_link_libraries(vulkanfix_driver_post PRIVATE ${CMAKE_DL_LIBS})
